%% Motor Digital Twin with Fault Injection and Residual-Based Detection
% Base MATLAB only (no toolboxes required).
%
% A DC motor "plant" (with parameter mismatch, sensor noise and an injected
% fault) is run under PI speed control. A healthy "digital twin" receives the
% same applied voltage. The difference between measured and predicted speed
% and current (the residual) reveals the fault.

clear; clc; close all; rng(3);
if ~exist('results','dir'), mkdir('results'); end

%% Nominal motor parameters (the digital twin uses these)
P.R = 1.2;      % armature resistance [ohm]
P.L = 2e-3;     % armature inductance [H]
P.Ke = 0.05;    % back-EMF constant [V/(rad/s)]
P.Kt = 0.05;    % torque constant   [N*m/A]
P.J = 1e-4;     % rotor inertia     [kg*m^2]
P.b = 1e-5;     % viscous friction  [N*m/(rad/s)]

%% Simulation settings
dt = 1e-4;  Tend = 3.0;  tFault = 1.5;  N = round(Tend/dt);
wRef = 150;                         % speed reference [rad/s]
Kp = 0.15;  Ki = 1.5;  Vmax = 24;   % PI controller and voltage limit
sigW = 0.5; sigI = 0.02;            % sensor noise std (speed, current)

faults = {'none','friction','resistance'};
res = struct();

for fi = 1:numel(faults)
    f = faults{fi};
    [t, w_meas, i_meas, w_twin, i_twin] = ...
        simulate(P, f, dt, N, tFault, wRef, Kp, Ki, Vmax, sigW, sigI);

    % Residuals: measured minus twin prediction
    rW = w_meas - w_twin;
    rI = i_meas - i_twin;

    % Moving RMS of residual (window = 50 ms)
    win = round(0.05/dt);
    sW = sqrt(movmean(rW.^2, [win-1 0]));   % causal window
    sI = sqrt(movmean(rI.^2, [win-1 0]));

    % Threshold learned from a healthy window (0.9 s to 1.45 s)
    hIdx = t >= 0.9 & t <= 1.45;
    thW = mean(sW(hIdx)) + 6*std(sW(hIdx));
    thI = mean(sI(hIdx)) + 6*std(sI(hIdx));

    alarm = (sW > thW) | (sI > thI);
    alarm(t < 0.9) = false;          % ignore start-up transient

    % Detection delay = first alarm after the fault time
    k = find(alarm & t >= tFault, 1, 'first');
    falseAlarm = any(alarm & t < tFault);
    if isempty(k)
        delay = NaN;
    else
        delay = t(k) - tFault;
    end

    res.(f) = struct('t',t,'w',w_meas,'wt',w_twin,'sW',sW,'sI',sI, ...
                     'thW',thW,'thI',thI,'alarm',alarm,'delay',delay, ...
                     'falseAlarm',falseAlarm);
    if strcmp(f,'none')
        fprintf('Fault: %-10s | false alarm before fault time: %d | alarms after: %d\n', ...
                f, falseAlarm, any(alarm & t >= tFault));
    else
        fprintf('Fault: %-10s | detected: %d | detection delay: %.1f ms | false alarm before fault: %d\n', ...
                f, ~isnan(delay), delay*1000, falseAlarm);
    end
end

%% Plots
for fi = 2:numel(faults)
    f = faults{fi}; r = res.(f);
    figure('Name', ['Fault: ' f], 'Position', [100 100 800 650]);
    subplot(3,1,1);
    plot(r.t, r.w, r.t, r.wt, 'LineWidth', 1); hold on;
    yl = ylim; plot([tFault tFault], yl, '--k');
    legend('Plant (measured)','Digital twin','Fault injected','Location','best');
    ylabel('Speed [rad/s]'); title(['Injected fault: ' f]); grid on;
    subplot(3,1,2);
    plot(r.t, r.sW, 'LineWidth', 1); hold on;
    plot([0 Tend], [r.thW r.thW], '--r');
    yl = ylim; plot([tFault tFault], yl, '--k');
    ylabel('Speed residual RMS'); grid on;
    subplot(3,1,3);
    plot(r.t, r.sI, 'LineWidth', 1); hold on;
    plot([0 Tend], [r.thI r.thI], '--r');
    yl = ylim; plot([tFault tFault], yl, '--k');
    ylabel('Current residual RMS'); xlabel('Time [s]'); grid on;
    saveas(gcf, fullfile('results', ['fault_' f '.png']));
end

%% ---------------- Local functions ----------------
function [t, w_meas, i_meas, w_twin, i_twin] = ...
        simulate(P, fault, dt, N, tFault, wRef, Kp, Ki, Vmax, sigW, sigI)

    % Plant = nominal parameters with 3% random mismatch
    Q = P;
    for nm = {'R','L','Ke','Kt','J','b'}
        Q.(nm{1}) = P.(nm{1}) * (1 + 0.03*(2*rand-1));
    end

    t = (0:N-1)'*dt;
    w = 0; i = 0;            % plant states
    wt = 0; it = 0;          % twin states
    integ = 0;
    w_meas = zeros(N,1); i_meas = zeros(N,1);
    w_twin = zeros(N,1); i_twin = zeros(N,1);

    for n = 1:N
        % Inject fault into the plant parameters
        Qn = Q;
        if t(n) >= tFault
            switch fault
                case 'friction',   Qn.b = 40*Q.b;      % bearing wear
                case 'resistance', Qn.R = 1.6*Q.R;     % winding degradation
            end
        end

        % Noisy measurements
        wm = w + sigW*randn;
        im = i + sigI*randn;
        w_meas(n) = wm; i_meas(n) = im;
        w_twin(n) = wt; i_twin(n) = it;

        % PI speed controller with anti-windup
        e = wRef - wm;
        Vc = Kp*e + Ki*integ;
        if abs(Vc) < Vmax, integ = integ + e*dt; end
        Vc = max(min(Vc, Vmax), -Vmax);

        % Plant dynamics (Euler)
        di = (Vc - Qn.R*i - Qn.Ke*w)/Qn.L;
        dw = (Qn.Kt*i - Qn.b*w)/Qn.J;
        i = i + di*dt;  w = w + dw*dt;

        % Digital twin: healthy nominal model, same voltage
        dit = (Vc - P.R*it - P.Ke*wt)/P.L;
        dwt = (P.Kt*it - P.b*wt)/P.J;
        it = it + dit*dt;  wt = wt + dwt*dt;
    end
end
